Web Modeler

Let's install web modeler UI.

Create a new database "web-modeler" on the current PostgreSQL instance

# Open a shell to the PostgreSQL pod - via GCP Cloud Shell or via Headlamp.
#
kubectl exec -it {{UNAMESPACE}}-postgresql-0 -n {{UNAMESPACE}} -- psql -U postgres
#
# If you are using Headlamp to open a shell, use the command: psql -U postgres

# You may have to enter the password for the postgres user.

# After you are logged in, run the following commands.
CREATE DATABASE "web-modeler";

CREATE USER "web-modeler" WITH PASSWORD 'makelabs';

GRANT ALL PRIVILEGES ON DATABASE "web-modeler" TO "web-modeler";

GRANT USAGE ON SCHEMA public TO "web-modeler";
GRANT CREATE ON SCHEMA public TO "web-modeler";
GRANT ALL PRIVILEGES ON DATABASE "web-modeler" TO "web-modeler";

ALTER DATABASE "web-modeler" OWNER TO "web-modeler";

exit;

Run helm upgrade to add Web Modeler UI


UNAMESPACE={{UNAMESPACE}} envsubst < values-combined-ingress.yaml > values-combined-ingress.my-ns.yaml

UNAMESPACE={{UNAMESPACE}} envsubst < values-connectors-env.yaml   > values-connectors-env.my-ns.yaml

UNAMESPACE={{UNAMESPACE}} envsubst < values-webmodeler.yaml   > values-webmodeler.my-ns.yaml

helm upgrade {{UNAMESPACE}} camunda/camunda-platform \
  -f values-combined-ingress.my-ns.yaml \
  -f values-connectors-env.my-ns.yaml \
  -f values-webmodeler.my-ns.yaml \
  --set zeebe.prometheusServiceMonitor.enabled=true \
  --version 14.4.1 --namespace {{UNAMESPACE}} --server-side=false

envsubst {{UNAMESPACE}} < c8-ingress-webmodeler.yaml | kubectl apply  -f -
